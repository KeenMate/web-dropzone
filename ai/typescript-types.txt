MULTISELECTOPTION INTERFACE
---------------------------
Standard option interface:

interface MultiSelectOption {
  value: string | number;         // Unique identifier
  label: string;                  // Display text
  icon?: string;                  // Icon or emoji
  subtitle?: string;              // Secondary text
  group?: string;                 // Group name
  disabled?: boolean;             // Disabled state
  selected?: boolean;             // Initial selection
}

Usage:
const options: MultiSelectOption[] = [
  { value: 1, label: 'JavaScript', icon: '🟨' },
  { value: 2, label: 'TypeScript', icon: '🔷' }
];

MULTISELECTEVENTDETAIL INTERFACE
--------------------------------
Event detail types:

interface SelectEventDetail<T = any> {
  option: T;                      // Selected option
  selectedOptions: T[];           // All selected options
}

interface DeselectEventDetail<T = any> {
  option: T;                      // Deselected option
  selectedOptions: T[];           // Remaining options
}

interface ChangeEventDetail<T = any> {
  selectedOptions: T[];           // All selected options
  selectedValues: (string | number)[]; // Values only
}

Usage:
select.addEventListener('change', (e: CustomEvent<ChangeEventDetail>) => {
  console.log(e.detail.selectedOptions);
  console.log(e.detail.selectedValues);
});

interface MultiSelectElement extends HTMLElement {

Usage:

if (select) {

}

GENERIC COMPONENT TYPE
----------------------
Type-safe component reference:

function setupMultiSelect<T extends MultiSelectOption>(
  selector: string,
  options: T[]
): MultiSelectElement {
  const select = document.querySelector(selector) as MultiSelectElement;

  select.options = options;

  select.changeCallback = (selected: T[]) => {
    console.log('Selected:', selected);
  };

  return select;
}

// Usage

  id: number;
  name: string;
  price: number;

const select = setupMultiSelect<Product>('web-multiselect', products);

TYPE GUARDS
-----------
Check types at runtime:

function isMultiSelectOption(obj: any): obj is MultiSelectOption {
  return (
    obj &&
    typeof obj === 'object' &&
    ('value' in obj) &&
    ('label' in obj)
  );
}

if (isMultiSelectOption(data)) {
  // TypeScript knows data is MultiSelectOption
  console.log(data.value, data.label);
}

FRAMEWORK INTEGRATION
---------------------
React with TypeScript:

import { useRef, useEffect } from 'react';
import type { MultiSelectElement, MultiSelectOption } from '@keenmate/web-multiselect';

interface Props {
  options: MultiSelectOption[];
  onChange: (selected: MultiSelectOption[]) => void;
}

function MyComponent({ options, onChange }: Props) {
  const selectRef = useRef<MultiSelectElement>(null);

  useEffect(() => {
    const select = selectRef.current;
    if (!select) return;

    select.options = options;

    const handleChange = (e: Event) => {
      const detail = (e as CustomEvent).detail;
      onChange(detail.selectedOptions);
    };

    select.addEventListener('change', handleChange);

    return () => {
      select.removeEventListener('change', handleChange);
    };
  }, [options, onChange]);

  return <web-multiselect ref={selectRef} />;
}

Vue 3 with TypeScript:

<script setup lang="ts">
import { ref, onMounted } from 'vue';
import type { MultiSelectElement, MultiSelectOption } from '@keenmate/web-multiselect';

interface Props {
  options: MultiSelectOption[];
}

const props = defineProps<Props>();
const select = ref<MultiSelectElement | null>(null);

onMounted(() => {
  if (select.value) {
    select.value.options = props.options;
  }
});
</script>

<template>
  <web-multiselect ref="select"></web-multiselect>
</template>

STRICT NULL CHECKS
------------------
Handle nullable types:

function getSelectValue(select: MultiSelectElement | null): (string | number)[] {
  if (!select) return [];

  const value = select.getValue();

  if (Array.isArray(value)) {
    return value;
  }

  return [value];
}
